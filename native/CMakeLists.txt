cmake_minimum_required(VERSION 3.22)
project(PianoPocket LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(BUILD_CXXLIBS OFF CACHE BOOL "" FORCE)
set(INSTALL_MANPAGES OFF CACHE BOOL "" FORCE)
set(WITH_OGG OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory(../third_party/flac flac EXCLUDE_FROM_ALL)
add_subdirectory(../third_party/lame lame)
add_library(pocket_vorbis STATIC ../third_party/stb/stb_vorbis.c)
target_compile_definitions(pocket_vorbis PRIVATE STB_VORBIS_MAX_CHANNELS=2)
add_library(pocket_core STATIC SampleDecoder.cpp SampleCache.cpp Effects.cpp SamplerEngine.cpp ImpulseLoader.cpp OfflineRenderer.cpp CompressedEncoder.cpp CompressedSampleDecoder.cpp)
target_link_libraries(pocket_core PUBLIC FLAC mp3lame pocket_vorbis)
target_include_directories(pocket_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(pocket_core PRIVATE -Wall -Wextra)
set_target_properties(pocket_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
  target_compile_options(pocket_core PRIVATE -O2)
  add_subdirectory(../third_party/oboe oboe)
  add_library(pianopocket SHARED AudioEngine.cpp JniBridge.cpp)
  target_link_libraries(pianopocket pocket_core oboe android log)
  target_link_options(pianopocket PRIVATE "-Wl,-z,max-page-size=16384")
else()
  enable_testing()
  find_package(Threads REQUIRED)
  target_link_libraries(pocket_core PUBLIC Threads::Threads)
  add_executable(native_tests ../tests/native/tests.cpp)
  target_link_libraries(native_tests pocket_core)
  add_test(NAME native_tests COMMAND native_tests)
  add_executable(codec_probe ../tests/native/codec_probe.cpp)
  target_link_libraries(codec_probe pocket_core)
  add_executable(sample_decode_probe ../tests/native/sample_decode_probe.cpp)
  target_link_libraries(sample_decode_probe pocket_core)
  add_executable(owned_preset_probe ../tests/native/owned_preset_probe.cpp)
  target_link_libraries(owned_preset_probe pocket_core)
endif()
